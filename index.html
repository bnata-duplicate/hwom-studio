-- ── M12: KPI TANIM TABLOSU ───────────────────────────── CREATE TABLE IF NOT EXISTS kpi_tanim ( id UUID DEFAULT gen_random_uuid() PRIMARY KEY, isletme_id UUID REFERENCES isletme(id) ON DELETE CASCADE, departman TEXT, kpi_adi TEXT NOT NULL, birim TEXT, formul TEXT, olcum_detayi TEXT, agirlik NUMERIC(4,2), kabul_sapma NUMERIC(4,2), hedef_2026 TEXT, veri_kaynagi TEXT, x_layer TEXT DEFAULT 'OX', olcum_sikligi TEXT DEFAULT 'aylik' CHECK (olcum_sikligi IN ('haftalik','aylik','ceyreklik','yillik')), kritik BOOLEAN DEFAULT false, aktif BOOLEAN DEFAULT true, olusturma_tarihi TIMESTAMPTZ DEFAULT NOW(), guncelleme_tarihi TIMESTAMPTZ DEFAULT NOW() ); -- ── M12: KPI GERÇEKLEŞEN TABLOSU ────────────────────── CREATE TABLE IF NOT EXISTS kpi_gerceklesen ( id UUID DEFAULT gen_random_uuid() PRIMARY KEY, isletme_id UUID REFERENCES isletme(id) ON DELETE CASCADE, kpi_id UUID REFERENCES kpi_tanim(id) ON DELETE CASCADE, gerceklesen_deger NUMERIC(12,4), olcum_tarihi DATE NOT NULL DEFAULT CURRENT_DATE, donem TEXT DEFAULT 'aylik', veri_kaynagi TEXT, not_aciklama TEXT, giris_yapan TEXT, olusturma_tarihi TIMESTAMPTZ DEFAULT NOW() ); -- ── OTOMATİK GÜNCELLEME TRİGGER ─────────────────────── CREATE OR REPLACE FUNCTION update_guncelleme_tarihi() RETURNS TRIGGER AS $$ BEGIN NEW.guncelleme_tarihi = NOW(); RETURN NEW; END; $$ LANGUAGE plpgsql; CREATE TRIGGER is_plani_guncelle BEFORE UPDATE ON is_plani FOR EACH ROW EXECUTE FUNCTION update_guncelleme_tarihi(); CREATE TRIGGER kpi_tanim_guncelle BEFORE UPDATE ON kpi_tanim FOR EACH ROW EXECUTE FUNCTION update_guncelleme_tarihi(); CREATE POLICY "kpi_gerceklesen_public" ON kpi_gerceklesen FOR ALL USING (true) WITH CHECK (true); -- ── KONTROL ──────────────────────────────────────────── SELECT table_name, (SELECT COUNT(*) FROM information_schema.columns WHERE table_name = t.table_name AND table_schema='public') AS kolon_sayisi FROM information_schema.tables t WHERE table_schema = 'public' AND table_name IN ('is_plani','kpi_tanim','kpi_gerceklesen','isletme','departman','okr','denetim') ORDER BY table_name;
